WITH audit_events AS (
  SELECT
    event_key,
    event_time,
    action,
    actor,
    actor_id,
    org,
    repository,
    ingested_at
  FROM
    `db-dev-l4to-mp-github.github_governance.github_audit_events_normalized`
  WHERE
    event_time >= TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 180 DAY)
    AND actor IS NOT NULL
),

deduplicated AS (
  SELECT * EXCEPT(row_num)
  FROM (
    SELECT
      *,
      ROW_NUMBER() OVER (
        PARTITION BY event_key
        ORDER BY ingested_at DESC
      ) AS row_num
    FROM audit_events
  )
  WHERE row_num = 1
),

user_activity AS (
  SELECT
    actor AS user_name,

    MAX(
      IF(
        action != 'business.sso_response',
        event_time,
        NULL
      )
    ) AS latest_activity,

    MAX(
      IF(
        action = 'business.sso_response',
        event_time,
        NULL
      )
    ) AS latest_authentication,

    MAX(
      IF(
        STARTS_WITH(action, 'git.'),
        event_time,
        NULL
      )
    ) AS latest_git_activity,

    COUNTIF(
      action != 'business.sso_response'
    ) AS activity_event_count_180d,

    COUNTIF(
      STARTS_WITH(action, 'git.')
    ) AS git_event_count_180d

  FROM deduplicated
  GROUP BY actor
)

SELECT
  user_name,
  latest_activity,
  latest_authentication,
  latest_git_activity,

  TIMESTAMP_DIFF(
    CURRENT_TIMESTAMP(),
    latest_activity,
    DAY
  ) AS days_since_activity,

  activity_event_count_180d,
  git_event_count_180d,

  CASE
    WHEN latest_activity >=
      TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 30 DAY)
      THEN 'ACTIVE_30D'

    WHEN latest_activity >=
      TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 45 DAY)
      THEN 'DORMANT_30D'

    WHEN latest_activity >=
      TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 60 DAY)
      THEN 'DORMANT_45D'

    WHEN latest_activity >=
      TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 90 DAY)
      THEN 'DORMANT_60D'

    ELSE 'DORMANT_90D_PLUS'
  END AS activity_status

FROM user_activity

ORDER BY latest_activity DESC;




SELECT
  MIN(event_time) AS earliest_event,
  MAX(event_time) AS latest_event,

  TIMESTAMP_DIFF(
    MAX(event_time),
    MIN(event_time),
    DAY
  ) AS available_history_days,

  COUNT(*) AS events,

  COUNT(DISTINCT DATE(event_time)) AS days_with_events

FROM
  `db-dev-l4to-mp-github.github_governance.github_audit_events_normalized`

WHERE
  event_time >= TIMESTAMP_SUB(
    CURRENT_TIMESTAMP(),
    INTERVAL 180 DAY
  );







WITH hours AS (
  SELECT hour
  FROM UNNEST(
    GENERATE_TIMESTAMP_ARRAY(
      TIMESTAMP_SUB(
        TIMESTAMP_TRUNC(CURRENT_TIMESTAMP(), HOUR),
        INTERVAL 90 DAY
      ),
      TIMESTAMP_TRUNC(CURRENT_TIMESTAMP(), HOUR),
      INTERVAL 1 HOUR
    )
  ) AS hour
),

events_by_hour AS (
  SELECT
    TIMESTAMP_TRUNC(event_time, HOUR) AS hour,
    COUNT(*) AS events
  FROM
    `db-dev-l4to-mp-github.github_governance.github_audit_events_normalized`
  WHERE
    event_time >= TIMESTAMP_SUB(
      CURRENT_TIMESTAMP(),
      INTERVAL 90 DAY
    )
  GROUP BY hour
)

SELECT
  h.hour,
  COALESCE(e.events, 0) AS events
FROM hours h
LEFT JOIN events_by_hour e USING(hour)
WHERE COALESCE(e.events, 0) = 0
ORDER BY h.hour;
